    "003": "잔디밭 벤치",
    "004": "빈 강의실",
    "005": "마이 교수 연구실",
    "006": "유저 자취방",
    "007": "대형 강의실"
  };
  const displayLocation = props.location_name || locationMap[resolvedLoc] || resolvedLoc || "도서관 열람실";
  const charName = props.char_name || "아야메";

  // 3. 개인(1:1) 루트 & 단체(올스타) 루트 겸용 동적 스탯 DB(roster_stats) 파싱
  // 프롤로그에서 주입된 활성 히로인만 표시하며, 미등장 히로인은 스포일러 없이 숨김 처리 후 조우 시 자동 해금!
  const rawRosterStr = props.roster_stats || gRoster || `${charName}:${props.affection || 30}/${props.stress || 15}`;
  const rosterMap = {};
  String(rawRosterStr).split(",").forEach((item) => {
    const parts = item.trim().split(":");
    if (parts.length === 2 && parts[0].trim()) {
      const cKey = parts[0].trim();
      const vals = parts[1].split("/");
      rosterMap[cKey] = {
        aff: Math.min(100, Math.max(0, parseInt(vals[0], 10) || 20)),
        stress: Math.min(100, Math.max(0, parseInt(vals[1], 10) || 15))
      };
    }
  });

  // 현재 대화 중인 캐릭터(charName)의 호감도·피로도 결정 (props 우선, 없으면 rosterMap에서 복원)
  const savedCharStat = rosterMap[charName] || { aff: 20, stress: 15 };
  const affNum = Math.min(100, Math.max(0, parseInt(props.affection !== undefined ? props.affection : savedCharStat.aff, 10)));
  const stressNum = Math.min(100, Math.max(0, parseInt(props.stress !== undefined ? props.stress : savedCharStat.stress, 10)));
  rosterMap[charName] = { aff: affNum, stress: stressNum };

  const serializeRoster = (mapObj) => {
    return Object.keys(mapObj)
      .map((k) => `${k}:${mapObj[k].aff}/${mapObj[k].stress}`)
      .join(",");
  };
  const updatedRosterStr = serializeRoster(rosterMap);
  const reaction = props.reaction || "상황을 지켜보고 있습니다...";

  // 4. 전역 변수 동기화 및 다음 턴 AI에게 반환변수(setReturnValue) 자동 전달
  const syncReturnValues = (nextTurn, nextTime, nextProg, nextDday, nextAff, nextStress, nextLoc, nextRoster) => {
    if (typeof setTemplateValue === "function") {
      setTemplateValue("turn", String(nextTurn));
      setTemplateValue("time", String(nextTime));
      setTemplateValue("prog", String(nextProg));
      setTemplateValue("d_day", String(nextDday));
      setTemplateValue("affection", String(nextAff));
      setTemplateValue("stress", String(nextStress));
      setTemplateValue("location", String(nextLoc));
      setTemplateValue("roster_stats", String(nextRoster));
    }
    if (typeof setReturnValue === "function") {
      setReturnValue("turn", nextTurn);
      setReturnValue("time", nextTime);
      setReturnValue("prog", nextProg);
      setReturnValue("d_day", nextDday);
      setReturnValue("affection", nextAff);
      setReturnValue("stress", nextStress);
      setReturnValue("location", nextLoc);
      setReturnValue("roster_stats", nextRoster);
    }
  };

  syncReturnValues(turnNum, resolvedTime, progNum, dDayNum, affNum, stressNum, resolvedLoc, updatedRosterStr);

  // 5. 호감도 4단계(Phase 1~4) 태도 & 약한 저항(1회 튕기기) UI 테마
  let phaseBadge = "🌱 Phase 1: 새침한 탐색 (1회 튕김)";
  let phaseGuide = "과한 스킨십엔 부끄러워 살짝 튕기지만, 한 번 더 다정히 다가가면 받아줍니다.";
  let phaseColor = "#64748b";
  let skinshipBtnText = "🤝 살짝 손끝 포개기 (호감+5)";
  let skinshipMsg = `[스킨십 시도] 부끄러워하며 살짝 빼려던 ${charName}의 손끝을 놓지 않고 다정하게 한 번 더 감싸 쥐며 눈을 맞춘다.`;
  let affBoost = 5;

  if (affNum >= 90) {
    phaseBadge = "💖 Phase 4: 완전한 헌신 (전속 연인)";
    phaseGuide = "완전 함락 상태! 모든 농밀한 스킨십이 허용되며 그녀가 먼저 당신을 갈망합니다.";
    phaseColor = "#db2777";
    skinshipBtnText = "❤️‍🔥 깊고 농밀하게 안아 올리기 (호감+8)";
    skinshipMsg = `[농밀한 밀착] ${charName}의 허리를 깊숙이 끌어안고 뜨거운 숨결이 닿을 듯 입술을 포갠다.`;
    affBoost = 8;
  } else if (affNum >= 70) {
    phaseBadge = "🔥 Phase 3: 뜨거운 밀착 (심야 해금)";
    phaseGuide = "거부할 수 없는 끌림! 키스와 단둘만의 공간(자취방·빈 강의실) 밀착이 해금됩니다.";
    phaseColor = "#ec4899";
    skinshipBtnText = "💋 단둘이 밀착해 분위기 잡기 (호감+7)";
    skinshipMsg = `[밀착 스킨십] ${charName}를 품 가까이로 끌어당기며 달아오른 뺨과 입술에 천천히 다가간다.`;
    affBoost = 7;
  } else if (affNum >= 40) {
    phaseBadge = "🌸 Phase 2: 설레는 호감 (스킨십 허용)";
    phaseGuide = "손잡기·머리 쓰다듬기를 기꺼이 반기며, 깊은 스킨십엔 귀엽게 살짝 앙탈을 부립니다.";
    phaseColor = "#f43f5e";
    skinshipBtnText = "💕 머리 쓰다듬으며 칭찬하기 (호감+6)";
    skinshipMsg = `[다정한 스킨십] ${charName}의 부드러운 머리칼을 쓰다듬으며 오늘 정말 예쁘고 든든하다고 칭찬해 준다.`;
    affBoost = 6;
  }

  // 6. 피로도(과제 시↑ / 놀기 시↓) 및 마감 위기 상태 판정
  const isMoodDown = stressNum >= 50;
  const isCrisis = dDayNum <= 7 && progNum < 50;

  // 7. 원클릭 인터랙션 퀵 액션 핸들러 (현재 캐릭터의 개별 스탯만 정확히 갱신하여 roster_stats에 저장)
  const handleQuickAction = (type) => {
    if (type === "skinship") {
      const nextAff = Math.min(100, affNum + affBoost);
      const nextStress = Math.max(0, stressNum - 8);
      rosterMap[charName] = { aff: nextAff, stress: nextStress };
      syncReturnValues(turnNum + 1, resolvedTime, progNum, dDayNum, nextAff, nextStress, resolvedLoc, serializeRoster(rosterMap));
      if (typeof sendMessage === "function") {
        sendMessage(skinshipMsg);
      }
    } else if (type === "play") {

      boxShadow: "0 4px 14px rgba(0, 0, 0, 0.08)",

      <div style={{
        display: "flex",
        justifyContent: "space-between",
        alignItems: "center",
        paddingBottom: "10px",
        marginBottom: "10px",
        borderBottom: "1px solid light-dark(#f4f4f5, #27272a)"
      }}>
        <div style={{ display: "flex", alignItems: "center", gap: "6px" }}>
          <span style={{
            fontSize: "11px",
            fontWeight: "800",
            padding: "3px 8px",
            borderRadius: "6px",
            background: isNight ? "#312e81" : "#fef3c7",
            color: isNight ? "#e0e7ff" : "#b45309"
          }}>
            {isNight ? "🌙 밤" : "☀️ 낮"}

            borderRadius: "6px",
            background: "light-dark(#f4f4f5, #27272a)",

            📍 {displayLocation}

        </div>
        <div style={{ display: "flex", alignItems: "center", gap: "6px" }}>
          <span style={{
            fontSize: "12px",
            fontWeight: "800",
            padding: "2px 8px",
            borderRadius: "6px",
            background: isCrisis ? "#dc2626" : "#fef2f2",
            color: isCrisis ? "#ffffff" : "#dc2626",
            border: "1px solid #fecaca"
          }}>
            ⏳ {dDayLabel}{dDayNum} {isCrisis ? "🚨합숙찬스" : ""}

        </span>
        <span style={{

          <div style={{
            width: `${stressNum}%`,
            height: "100%",
            background: isMoodDown ? "linear-gradient(90deg, #fbbf24, #f59e0b)" : "linear-gradient(90deg, #34d399, #10b981)",
            transition: "width 0.3s ease"
          }} />
        </div>
      </div>

      {/* 스마트 멀티 스탯 보드: 단체(올스타) 루트이거나 2명 이상 조우했을 때만 조원/인연 현황 표시 */}
      {Object.keys(rosterMap).length > 1 && (
        <div style={{
          display: "flex",
          flexWrap: "wrap",
          alignItems: "center",
          gap: "4px",
          marginBottom: "10px",
          padding: "6px 8px",
          borderRadius: "8px",
          background: "light-dark(#f4f4f5, #27272a)"
        }}>
          <span style={{ fontSize: "10px", fontWeight: "700", color: "light-dark(#71717a, #a1a1aa)", marginRight: "2px" }}>
            👥 인연 현황:
          </span>
          {Object.keys(rosterMap).map((name) => {
            const st = rosterMap[name];
            const isCurrent = name === charName;
            return (
              <span
                key={name}
                style={{
                  fontSize: "10px",
                  fontWeight: isCurrent ? "800" : "600",
                  padding: "2px 6px",
                  borderRadius: "4px",
                  background: isCurrent ? phaseColor : "light-dark(#ffffff, #18181b)",
                  color: isCurrent ? "#ffffff" : "light-dark(#3f3f46, #d4d4d8)"
                }}
              >
                {name} 💖{st.aff}/☕{st.stress}
              </span>
            );
          })}
        </div>
      )}

      {/* 캐릭터 속마음 리액션 말풍선 */}
      <div style={{
        padding: "8px 10px",
        borderRadius: "8px",
        background: "light-dark(#f8fafc, #27272a)",
        fontSize: "12px",
        fontStyle: "italic",
        color: "light-dark(#000000, #ffffff)",
        fontWeight: "600",
        marginBottom: "10px",
        borderLeft: `3px solid ${phaseColor}`
      }}>
        "{reaction}"
      </div>

      {/* 호감도·피로도 연동 동적 퀵 액션 버튼 2종 */}
      <div style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: "6px" }}>
        <button
          onClick={() => handleQuickAction("skinship")}
          style={{
            padding: "8px 4px",
            borderRadius: "8px",
            border: `1px solid ${phaseColor}66`,
            background: "light-dark(#fdf2f8, #500724)",
            color: "light-dark(#000000, #ffffff)",
            fontSize: "11px",
            fontWeight: "700",
            cursor: "pointer"
          }}
        >
          {skinshipBtnText}
        </button>
        <button
          onClick={() => handleQuickAction("play")}
          style={{
            padding: "8px 4px",
            borderRadius: "8px",
            border: isMoodDown ? "1px solid #f59e0b" : "1px solid light-dark(#e2e8f0, #334155)",
            background: isMoodDown ? "light-dark(#fffbeb, #451a03)" : "light-dark(#f1f5f9, #1e293b)",
            color: "light-dark(#000000, #ffffff)",
            fontSize: "11px",
            fontWeight: "700",
            cursor: "pointer"
          }}
        >
          {isMoodDown ? "🍰 놀아주며 기분UP! (피로-22)" : "☕ 쉬어가며 수다 떨기 (피로-15)"}
        </button>
      </div>
    </div>
  );
}
